** setup.el
#+BEGIN_SRC emacs-lisp
(eval-when-compile
  (require 'setup)
  (require 'cl-lib)

  ;; From https://www.emacswiki.org/emacs/SetupEl#h5o-4
  (defmacro defsetup (name signature &rest body)
    "Shorthand for `setup-define'.
NAME is the name of the local macro.  SIGNATURE is used as the
argument list for FN.  If BODY starts with a string, use this as
the value for :documentation.  Any following keywords are passed
as OPTS to `setup-define'."
    (declare (debug defun))
    (let (opts)
      (when (stringp (car body))
        (setq opts (nconc (list :documentation (pop body)) opts)))
      (while (keywordp (car body))
        (let* ((prop (pop body))
               (val `',(pop body)))
          (setq opts (nconc (list prop val) opts))))
      `(setup-define ,name
         (cl-function (lambda ,signature ,@body))
         ,@opts)))

  ;; The following are adapted from akirak/emacs-config
  ;; (MIT, Copyright (C) 2023-2025 Akira Komamura)
  ;; https://github.com/akirak/emacs-config/blob/master/emacs-config.org
  (defsetup :package (package)
    "Fake installation of PACKAGE."
    :repeatable t
    :shorthand cadr)

  (defsetup :repeat (keymap &rest commands)
    "Set the repeat map of commands."
    :indent 1
    (dolist (cmd commands)
      (put cmd 'repeat-map keymap))))
#+END_SRC

** enlight
#+BEGIN_SRC emacs-lisp
(setup (:package enlight)
  (:require enlight)
  (setup (:package grid))
  (:require grid)
  (require 'bookmark)
  (require 'project)
  (setq inhibit-startup-message t)
  ;; We do all centering ourselves below; enlight's own centering measures
  ;; the banner image (a single character with a `display' override) as
  ;; 1 column wide / 1 line tall, ignoring its actual pixel dimensions.
  (setopt enlight-center-horizontally nil)
  (setopt enlight-center-vertically nil)

  (defvar my/enlight-footer-messages
    '("「君らにはもう 二度と危険ってやつは訪れない」"
      "「ちゃんとしてますね 実に」"
      "「SCIENCE IS ELEGANT」"
      "「はっは──!! 飛び出たぜついに!!! 宇宙は俺のものだ!!!!!」"
      "「……いや できるか?」「ああ できるね」"
      "「スイカがゼロから 科学をやれば……!!!」"
      "「たった 7年か 早かったな……!!」"))

  ;; Picked once per Emacs session (not on every resize-triggered rebuild).
  (defvar my/enlight-footer-message nil)

  (defun my/enlight-build-banner ()
    (let ((img-file (expand-file-name "assets/banner.png" user-emacs-directory)))
      (when (and (display-graphic-p) (image-type-available-p 'png) (file-exists-p img-file))
        ;; Use a non-whitespace placeholder char: grid.el's alignment logic
        ;; calls `delete-horizontal-space', which strips literal space
        ;; characters based on the underlying buffer text, ignoring any
        ;; `display' override, and would eat a plain " " placeholder.
        (propertize "." 'display (create-image img-file)))))

  (defun my/enlight-image-extra-lines (banner)
    "Extra lines BANNER's image occupies beyond its own single logical line."
    (if-let* ((img (and banner (get-text-property 0 'display banner))))
        (max 0 (1- (ceiling (cdr (image-size img t)) (frame-char-height))))
      0))

  (defun my/enlight-build-content ()
    ;; A couple of columns/lines of slack: at build time the selected window
    ;; may not yet be *enlight*'s own (e.g. it still has line numbers on),
    ;; so its reported size can be slightly off from what *enlight* ends up
    ;; with, causing the widest line to wrap by a column or two.
    (let* ((width (- (window-body-width (selected-window)) 3))
           (height (window-body-height (selected-window)))
           (banner (my/enlight-build-banner))
           (sections
            (seq-remove
             (lambda (section) (null (cdr section)))
             `(("Recent Files"
                ,@(cl-loop for file in (seq-take recentf-list 5)
                           for i from 1
                           collect (list (abbreviate-file-name file)
                                         `(find-file ,file)
                                         (number-to-string i))))
               ("Bookmarks"
                ,@(cl-loop for name in (seq-take (bookmark-all-names) 5)
                           collect (list name `(bookmark-jump ,name))))
               ("Projects"
                ,@(cl-loop for root in (seq-take (project-known-project-roots) 5)
                           collect (list (abbreviate-file-name root) `(dired ,root)))))))
           (body
            (grid-make-column
             (delq
              nil
              (list
               (when banner
                 (list :content banner :width width :align 'center :margin '(0 0 1 0)))
               (list :content (propertize "SUPER DUPER AMANATSU-CHAN!" 'face 'bold)
                     :width width :align 'center :margin '(0 0 1 0))
               (when sections
                 (list :content (enlight-menu sections)
                       :width width :align 'center :margin '(0 0 1 0)))
               (list :content (propertize
                               (or my/enlight-footer-message
                                   (setq my/enlight-footer-message
                                         (nth (random (length my/enlight-footer-messages))
                                              my/enlight-footer-messages)))
                               'face 'italic)
                     :width width :align 'center)))))
           (body-lines (with-temp-buffer (insert body) (count-lines (point-min) (point-max))))
           (top-pad (max 0 (/ (- height body-lines (my/enlight-image-extra-lines banner)) 2))))
      (concat (make-string top-pad ?\n) body)))

  (defun my/enlight-refresh ()
    (customize-set-variable 'enlight-content (my/enlight-build-content)))

  (defun my/enlight-rerender (&optional frame)
    (when (get-buffer-window "*enlight*" (or frame t))
      (my/enlight-refresh)
      (enlight)))

  (defun my/enlight-show ()
    ;; `enlight' computes vertical centering from the *enlight* window's
    ;; height, but that window doesn't exist yet on the first call (the
    ;; buffer isn't shown anywhere until `enlight-open' below pops it up).
    ;; Re-render once it actually has a window to measure.
    (enlight-open)
    (my/enlight-rerender))

  (add-hook 'after-init-hook #'my/enlight-refresh)
  (add-hook 'emacs-startup-hook #'my/enlight-show)
  (add-hook 'window-size-change-functions #'my/enlight-rerender)
  (add-hook 'enlight-mode-hook (lambda () (display-line-numbers-mode -1))))
#+END_SRC

*** urls-cleanup-string
#+BEGIN_SRC emacs-lisp
(defun azlle/urls-cleanup-string (str)
  "文字列STRに含まれるYouTube/X/TwitchのURLを整形して返す。"
  (let ((yt-re (rx "http" (opt "s") "://www.youtube.com/"
                   (or (seq "live/" (group (one-or-more (not (any "&? \n]")))))
                       (seq "watch?v=" (group (one-or-more (not (any "&? \n]"))))))
                   (opt (minimal-match (zero-or-more any)) (any "?&") "t="
                        (group (one-or-more (not (any "& \n]")))))
                   (zero-or-more (not (any " \n]")))))
        (x-re (rx (group "http" (opt "s") "://" (or "x" "twitter") ".com/"
                         (one-or-more (not (any "/ \n]"))) "/status/" (one-or-more digit))
                  (zero-or-more (not (any " \n]")))))
        (tw-re (rx "http" (opt "s") "://www.twitch.tv/"
                   (or (seq "videos/" (group (one-or-more digit)))
                       (seq (one-or-more (not (any "/ \n]"))) "/video/" (group (one-or-more digit))))
                   ;; "?" の後に "&区切りで t= が出るまで読み飛ばす" 形式に変更
                   (opt "?" (zero-or-more (seq (one-or-more (not (any "=& \n]"))) "="
                                               (zero-or-more (not (any "& \n]"))) "&"))
                        "t=" (group (one-or-more (not (any "& \n]")))))
                   (zero-or-more (not (any " \n]"))))))
    (cond
     ((string-match yt-re str)
      (let ((id (or (match-string 1 str) (match-string 2 str)))
            (time (match-string 3 str)))
        (if time
            (format "https://youtu.be/%s?t=%s" id time)
          (format "https://youtu.be/%s" id))))
     ((string-match x-re str) (match-string 1 str))
     ((string-match tw-re str)
      (let ((id (or (match-string 1 str) (match-string 2 str)))
            (time (match-string 3 str)))
        (if time
            (format "https://www.twitch.tv/videos/%s?t=%s" id time)
          (format "https://www.twitch.tv/videos/%s" id))))
     (t str))))
#+END_SRC

*** urls-cleanup
#+BEGIN_SRC emacs-lisp
(defun azlle/urls-cleanup ()
  "選択範囲またはバッファ全体のURLを一括整形する。"
  (interactive)
  (let ((beg (if (use-region-p) (region-beginning) (point-min)))
        (end (copy-marker (if (use-region-p) (region-end) (point-max)))))
    (save-excursion
      (goto-char beg)
      (let ((combined-re (rx (or (seq "http" (opt "s") "://www.youtube.com/" (or "live/" "watch?v="))
                                 (seq "http" (opt "s") "://" (or "x" "twitter") ".com/")
                                 ;; /videos/ と /CHANNEL/video/ の両方をキャッチ
                                 (seq "http" (opt "s") "://www.twitch.tv/"
                                      (or "videos/"
                                          (seq (one-or-more (not (any "/ \n]"))) "/video/"))))
                             (one-or-more (not (any " \n]"))))))
        (while (re-search-forward combined-re end t)
          (replace-match (azlle/urls-cleanup-string (match-string 0)) t t))))
    (set-marker end nil)
    (message "URLs cleaned!")))
